import React from "react";
import MaterialTable, { MTableToolbar } from "@material-table/core";
import TableFooter from "@material-table/core";
import TableCell from "@material-table/core";
import TableRow from "@material-table/core";
import { ExportCsv, ExportPdf } from "@material-table/exporters";

/**
 * @description Componente que muestra un informe de la colección de coches en una tabla.
 * Permite filtrar, exportar a CSV y PDF, y calcular la suma total de los precios.
 * @param {object} props Las propiedades del componente.
 * @param {array} props.datos Los datos de la colección de coches.
 */
function InformeColeccion(props) {
  /**
   * @description Definición de las columnas de la tabla.
   */
  const col = [
    { title: "Nombre", field: "firstName", filtering: false },
    { title: "Marca", field: "marca" },
    { title: "Tipo", field: "type" },
    { title: "Precio", field: "price", type: "numeric", filtering: false }
  ];

  /**
   * @description Transforma los datos de entrada al formato requerido por MaterialTable.
   */
  const tableData = props.datos.map((item) => ({
    id: item.id,
    firstName: item.nombre,
    marca: item.marca,
    type: item.tipo,
    price: item.precio,
  }));

  return (
    <div>
      <MaterialTable
        title="Coleccion de coches"
        columns={col}
        data={tableData}
        components={{
          /**
           * @description Personaliza la barra de herramientas de la tabla.
           * @param {object} props Las propiedades de la barra de herramientas.
           */
          Toolbar: (props) => (
            <div style={{ backgroundColor: "#808080" }}>
              <MTableToolbar {...props} />
            </div>
          ),
          /**
           * @description Personaliza el pie de la tabla para mostrar la suma total de los precios.
           * @param {object} props Las propiedades del pie de la tabla.
           */
          Footer: (props) => {
            const { data, renderSummaryRow } = props;
            const summaryRow = renderSummaryRow({ column: col[3], data }); // Sumar precios
            return (
              <TableFooter>
                <TableRow>
                  <TableCell colSpan={col.length} style={summaryRow?.style}>
                    Suma total de precios: {summaryRow?.value}
                  </TableCell>
                </TableRow>
              </TableFooter>
            );
          },
        }}
        options={{
          columnsButton: true,
          filtering: true,
          exportMenu: [
            {
              label: "Export PDF",
              exportFunc: (cols, datas) => ExportPdf(cols, datas, "ColeccionPDF"),
            },
            {
              label: "Export CSV",
              exportFunc: (cols, datas) => ExportCsv(cols, datas, "ColeccionCSV"),
            },
          ],
          headerStyle: {
            backgroundColor: '#000',
            color: '#FFF'
          },
          cellStyle: {
            backgroundColor: '#808080',
            color: '#FFF'
          }
        }}
        /**
         * @description Calcula la suma total de los precios.
         * @param {object} params Parámetros para renderizar la fila de resumen.
         * @returns {object} Objeto con el valor y estilo para la celda de resumen, o undefined si no es la columna de precio.
         */
        renderSummaryRow={({ column, data }) =>
          column.field === "price"
            ? {
              value: data.reduce((agg, row) => agg + row.price, 0),
              style: { background: "#333" },
            }
            : undefined
        }
      />
    </div>
  );
}

export default InformeColeccion;
