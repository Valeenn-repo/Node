import React from "react";
import MaterialTable, { MTableToolbar } from "@material-table/core";
import { ExportCsv, ExportPdf } from "@material-table/exporters";

/**
 * @description Componente que muestra un informe de usuarios en una tabla.
 * Permite filtrar y exportar a CSV y PDF.
 * @param {object} props Las propiedades del componente.
 * @param {array} props.datos Los datos de los usuarios.
 */
function InformeUsuarios(props) {
  /**
   * @description Definición de las columnas de la tabla.
   */
  const col = [
    { title: "Nombre", field: "firstName" },
    { title: "Login", field: "login", filtering: false },
    { title: "Password", field: "password", filtering: false },
    { title: "Rol", field: "rol", filtering: false }
  ];

  /**
   * @description Transforma los datos de entrada al formato requerido por MaterialTable.
   */
  const tableData = props.datos.map((item) => ({
    id: item.id,
    firstName: item.nombre,
    login: item.login,
    password: item.password,
    rol: item.rol,
  }));

  return (
    <div>
      <MaterialTable
        title="Usuarios"
        columns={col}
        data={tableData}
        components={{
          /**
           * @description Personaliza la barra de herramientas de la tabla.
           * @param {object} props Las propiedades de la barra de herramientas.
           */
          Toolbar: (props) => (
            <div style={{ backgroundColor: "#808080" }}>
              <MTableToolbar {...props} />
            </div>
          ),
        }}
        options={{
          columnsButton: true,
          filtering: true,
          exportMenu: [
            {
              label: "Export PDF",
              exportFunc: (cols, datas) => ExportPdf(cols, datas, "UsuariosPDF"),
            },
            {
              label: "Export CSV",
              exportFunc: (cols, datas) => ExportCsv(cols, datas, "UsuariosCSV"),
            },
          ],
          headerStyle: {
            backgroundColor: '#000',
            color: '#FFF'
          },
          cellStyle: {
            backgroundColor: '#808080',
            color: '#FFF'
          }
        }}
      />
    </div>
  );
}

export default InformeUsuarios;
